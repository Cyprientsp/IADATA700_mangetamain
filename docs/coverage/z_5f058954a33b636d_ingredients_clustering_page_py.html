212 def render_cooccurrence_analysis(self, ingredient_names: list[str], ingredients_matrix: pd.DataFrame) -> None: 

213 """Affiche l'analyse de co-occurrence interactive. 

214 

215 Permet à l'utilisateur de sélectionner deux ingrédients et visualise 

216 leur score de co-occurrence (nombre de recettes où ils apparaissent ensemble). 

217 Affiche également des statistiques contextuelles pour interpréter le score. 

218 

219 Args: 

220 ingredient_names: Liste des noms d'ingrédients disponibles pour la sélection. 

221 ingredients_matrix: Matrice de co-occurrence (DataFrame symétrique) où 

222 matrix[ing1, ing2] = nombre de recettes contenant ing1 ET ing2. 

223 

224 Raises: 

225 ValueError: Si les ingrédients sélectionnés ne sont pas dans la matrice. 

226 IndexError: Si un accès invalide à la matrice est tenté. 

227 """ 

228 st.subheader("🔍 Analyse de Co-occurrence") 

229 

230 # Création de trois colonnes pour les menus déroulants 

231 col1, col2, col3 = st.columns([1, 1, 1]) 

232 

233 with col1: 

234 ingredient1 = st.selectbox( 

235 "Premier ingrédient", 

236 options=ingredient_names, 

237 index=0, 

238 key="ingredient1", 

239 ) 

240 

241 with col2: 

242 ingredient2 = st.selectbox( 

243 "Deuxième ingrédient", 

244 options=ingredient_names, 

245 index=1 if len(ingredient_names) > 1 else 0, 

246 key="ingredient2", 

247 ) 

248 

249 # Afficher le score de co-occurrence 

250 if ingredient1 and ingredient2 and ingredient1 != ingredient2: 

251 try: 

252 # Récupérer le score de co-occurrence 

253 cooccurrence_score = ingredients_matrix.at[ingredient1, ingredient2] 

254 

255 # Calculer les statistiques de la matrice 

256 matrix_values = ingredients_matrix.values 

257 matrix_values_flat = matrix_values[matrix_values > 0] # Seulement les valeurs non-nulles 

258 

259 if len(matrix_values_flat) > 0: 

260 max_score = np.max(matrix_values_flat) 

261 avg_score = np.mean(matrix_values_flat) 

262 median_score = np.median(matrix_values_flat) 

263 else: 

264 max_score = avg_score = median_score = 0 

265 

266 # Affichage des métriques 

267 col_metric1, col_metric2, col_metric3 = st.columns(3) 

268 

269 with col_metric1: 

270 st.metric( 

271 label="Score de co-occurrence", 

272 value=f"{cooccurrence_score:.0f}", 

273 help=f"Nombre de recettes contenant '{ingredient1}' ET '{ingredient2}'", 

274 ) 

275 

276 with col_metric2: 

277 if max_score > 0: 

278 percentile = (cooccurrence_score / max_score) * 100 

279 st.metric( 

280 label="Percentile", 

281 value=f"{percentile:.1f}%", 

282 help="Position par rapport au score maximum", 

283 ) 

284 

285 with col_metric3: 

286 if avg_score > 0: 

287 ratio_avg = cooccurrence_score / avg_score 

288 st.metric( 

289 label="Ratio vs Moyenne", 

290 value=f"{ratio_avg:.1f}x", 

291 help=f"Ratio par rapport à la moyenne ({avg_score:.1f})", 

292 ) 

293 

294 # Barre de progression visuelle 

295 if max_score > 0: 

296 normalized_score = cooccurrence_score / max_score 

297 st.progress(normalized_score) 

298 

299 # Interprétation du score 

300 if cooccurrence_score >= median_score * 2: 

301 st.success("🔥 Combinaison très fréquente!") 

302 elif cooccurrence_score >= median_score: 

303 st.info("✅ Combinaison courante") 

304 elif cooccurrence_score > 0: 

305 st.warning("⚠️ Combinaison rare") 

306 else: 

307 st.error("❌ Aucune co-occurrence trouvée") 

308 

309 except (ValueError, IndexError, KeyError): 

310 st.warning("Erreur lors du calcul du score de co-occurrence")

350 def render_tsne_visualization(self, analyzer: IngredientsAnalyzer, clusters: np.ndarray, tsne_perplexity: int) -> None: 

351 """Affiche la visualisation t-SNE 2D des clusters d'ingrédients. 

352 

353 Génère et affiche un graphique interactif Plotly montrant les ingrédients 

354 dans un espace 2D obtenu par réduction de dimensionnalité t-SNE. Les points 

355 sont colorés selon leur cluster et peuvent être régénérés avec de nouveaux 

356 paramètres. 

357 

358 Args: 

359 analyzer: Instance de IngredientsAnalyzer utilisée pour générer la 

360 visualisation t-SNE à partir de la matrice de co-occurrence. 

361 clusters: Array numpy des labels de cluster pour chaque ingrédient. 

362 tsne_perplexity: Paramètre de perplexité pour t-SNE (5-50). 

363 Contrôle la densité des groupes dans la visualisation. 

364 Valeurs faibles = focus local, valeurs élevées = structure globale. 

365 

366 Notes: 

367 La visualisation est mise en cache dans st.session_state pour éviter 

368 de la recalculer à chaque interaction. Un bouton permet de forcer la 

369 régénération avec de nouveaux paramètres aléatoires. 

370 """ 

371 col_title, col_button = st.columns([3, 1]) 

372 with col_title: 

373 st.subheader("🎨 Visualisation t-SNE 2D des Clusters") 

374 with col_button: 

375 regenerate_tsne = st.button( 

376 "🔄 Régénérer t-SNE", 

377 help="Regénérer la visualisation avec de nouveaux paramètres", 

378 ) 

379 

380 # Générer t-SNE au premier lancement ou si demandé 

381 should_generate_tsne = "tsne_data" not in st.session_state or regenerate_tsne 

382 

383 if should_generate_tsne: 

384 with st.spinner("Génération de la visualisation t-SNE..."): 

385 tsne_data = analyzer.generate_tsne_visualization(clusters, perplexity=tsne_perplexity) 

386 st.session_state["tsne_data"] = tsne_data 

387 else: 

388 tsne_data = st.session_state["tsne_data"] 

389 

390 if "error" not in tsne_data: 

391 # Créer le graphique de dispersion avec Plotly 

392 fig_tsne = go.Figure() 

393 

394 # Palette de couleurs hexadécimales pour t-SNE 

395 tsne_colors = [ 

396 "#FF6B6B", 

397 "#4ECDC4", 

398 "#45B7D1", 

399 "#96CEB4", 

400 "#FFEAA7", 

401 "#DDA0DD", 

402 "#98D8C8", 

403 "#F7DC6F", 

404 "#BB8FCE", 

405 "#85C1E9", 

406 ] 

407 

408 n_clusters = tsne_data["n_clusters"] 

409 

410 # Ajouter les points par cluster pour avoir des couleurs distinctes 

411 for cluster_id in range(n_clusters): 

412 # Filtrer les données pour ce cluster 

413 cluster_mask = [label == cluster_id for label in tsne_data["cluster_labels"]] 

414 cluster_x = [x for i, x in enumerate(tsne_data["x_coords"]) if cluster_mask[i]] 

415 cluster_y = [y for i, y in enumerate(tsne_data["y_coords"]) if cluster_mask[i]] 

416 cluster_names = [name for i, name in enumerate(tsne_data["ingredient_names"]) if cluster_mask[i]] 

417 

418 color = tsne_colors[cluster_id % len(tsne_colors)] 

419 

420 fig_tsne.add_trace( 

421 go.Scatter( 

422 x=cluster_x, 

423 y=cluster_y, 

424 mode="markers+text", 

425 marker=dict( 

426 size=12, 

427 color=color, 

428 line=dict(width=2, color="white"), 

429 opacity=0.8, 

430 ), 

431 text=cluster_names, 

432 textposition="top center", 

433 textfont=dict(size=10), 

434 name=f"Cluster {cluster_id + 1}", 

435 hovertemplate="<b>%{text}</b><br>Cluster: " 

436 + f"{cluster_id + 1}<br>" 

437 + "Coordonnées: (%{x:.2f}, %{y:.2f})<extra></extra>", 

438 ) 

439 ) 

440 

441 # Mise en forme du graphique 

442 fig_tsne.update_layout( 

443 title="Visualisation t-SNE des Ingrédients par Cluster", 

444 xaxis_title="Dimension t-SNE 1", 

445 yaxis_title="Dimension t-SNE 2", 

446 showlegend=True, 

447 height=600, 

448 hovermode="closest", 

449 plot_bgcolor="rgba(245,245,245,0.8)", 

450 legend=dict(orientation="h", yanchor="bottom", y=1.02, xanchor="right", x=1), 

451 ) 

452 

453 # Afficher le graphique 

454 st.plotly_chart(fig_tsne, use_container_width=True) 

455 

456 # Informations sur t-SNE 

457 with st.expander("ℹ️ À propos de la visualisation t-SNE / Diagnostics"): 

458 st.markdown( 

459 """ 

460 **t-SNE (t-Distributed Stochastic Neighbor Embedding)** est une technique de réduction de dimensionnalité 

461 qui permet de visualiser des données haute-dimensionnelles en 2D. 

462 

463 **Dans ce contexte :** 

464 - Chaque point représente un **ingrédient** 

465 - La position est basée sur les **profils de co-occurrence** avec les autres ingrédients 

466 - Les couleurs correspondent aux **clusters K-means** 

467 - Les ingrédients proches ont des **patterns de co-occurrence similaires** 

468 

469 **Interprétation :** 

470 - Points regroupés = ingrédients utilisés dans des contextes similaires 

471 - Clusters colorés = groupes détectés par l'algorithme K-means 

472 - Distance = mesure de similarité des profils culinaires 

473 """ 

474 ) 

475 

476 method = tsne_data.get("tsne_params", {}).get("method", "tsne") 

477 st.markdown( 

478 f""" 

479 **Paramètres & Méthode :** 

480 - Méthode effective : `{method}` 

481 - Perplexité (après ajustement) : {tsne_data['tsne_params']['perplexity']} 

482 - Itérations max : {tsne_data['tsne_params']['max_iter']} 

483 - Seed aléatoire : {tsne_data['tsne_params']['random_state']} 

484 - Ingrédients (n_samples) : {len(tsne_data['ingredient_names'])} 

485 """ 

486 ) 

487 

488 if method != "tsne": 

489 if method == "fallback_circle": 

490 st.warning( 

491 "Fallback circle layout utilisé car t-SNE instable (trop peu d'ingrédients ou matrice dégénérée)." 

492 ) 

493 elif method == "fallback_svd": 

494 st.info("Projection issue de la décomposition SVD (approximation PCA) suite à un échec t-SNE.") 

495 

496 # Afficher quelques stats basiques sur la dispersion 

497 try: 

498 xs = tsne_data["x_coords"] 

499 ys = tsne_data["y_coords"] 

500 spread_x = max(xs) - min(xs) 

501 spread_y = max(ys) - min(ys) 

502 st.caption(f"Dispersion: Δx={spread_x:.2f}, Δy={spread_y:.2f} (échelle relative des clusters)") 

503 except Exception: 

504 pass 

505 else: 

506 st.error("Erreur lors de la génération de la visualisation t-SNE") 

507 with st.expander("🛠 Détails de l'erreur"): 

508 st.json(tsne_data) 

509 st.markdown( 

510 """ 

511 **Causes possibles :** 

512 - Perplexité trop élevée par rapport au nombre d'ingrédients (doit être < n_samples) 

513 - Matrice de co-occurrence vide ou dégénérée (toutes valeurs nulles) 

514 - Incohérence entre le nombre de labels de clusters et la liste d'ingrédients 

515 - Conflit de cache sur des anciennes données 

516 

517 **Actions suggérées :** 

518 1. Réduire le nombre d'ingrédients ou ajuster la perplexité 

519 2. Vider le cache (bouton Clear Cache) puis relancer 

520 3. Vérifier que l'étape de clustering a bien été effectuée 

521 """ 

522 )

525 """Affiche les statistiques de clustering dans la sidebar. 

526 

527 Présente des métriques récapitulatives et un graphique de répartition

529 disponibles. 

530 

531 Args: 

532 clusters: Array numpy des labels de cluster, ou None si l'analyse

534 ingredient_names: Liste des noms d'ingrédients, ou None si l'analyse 

535 n'a pas encore été effectuée. 

536 

537 Notes: 

538 Cette méthode vérifie que les deux paramètres sont non-None avant 

539 d'afficher les statistiques. Le graphique utilise Plotly pour une 

540 visualisation interactive. 

541 """ 

542 if clusters is not None and ingredient_names is not None: 

543 st.sidebar.markdown("---") 

544 st.sidebar.markdown("### 📊 Statistiques") 

545 

546 # Comptage par cluster 

547 cluster_counts = pd.Series(clusters).value_counts().sort_index() 

548 

549 st.sidebar.metric("Total ingrédients", len(ingredient_names)) 

550 st.sidebar.metric("Nombre de clusters", len(cluster_counts)) 

551 

552 # Graphique horizontal des proportions par cluster 

553 st.sidebar.markdown("**Répartition par cluster:**") 

554 

555 # Créer le graphique avec Plotly 

556 colors = [ 

557 "#FF6B6B", 

558 "#4ECDC4", 

559 "#45B7D1", 

560 "#96CEB4", 

561 "#FFEAA7", 

562 "#DDA0DD", 

563 "#98D8C8", 

564 "#F7DC6F", 

565 "#BB8FCE",

569 fig = go.Figure() 

570
